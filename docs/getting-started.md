# Build your first crawler

[Documentation index](README.md) · Next: [Writing site crawlers](crawlers.md)

This example collects page titles from a site you choose. It includes a minimal
domain so you can run it without an existing domain app. Run commands from the
new app directory throughout this guide.

## 1. Create the app

Use Node.js 26.13.1 or a compatible Node 26 release and npm 11.20 or later. The
example uses CommonJS TypeScript because jscrawl loads crawler modules through
`require` and `ts-node/register`.

```sh
mkdir my-crawler
cd my-crawler
npm init -y
npm install jscrawl gumbo-html sqlex zod
npm install --save-dev typescript ts-node @types/node
mkdir -p src/mcp etc crawlers/fixtures/example
```

Add these scripts to `package.json`, keeping its other fields. Leave `type`
unset or set it to `commonjs`.

```json
{
  "scripts": {
    "jscrawl": "node -r ts-node/register src/cli.ts",
    "start": "node -r ts-node/register src/server.ts",
    "db:schema": "node -r ts-node/register src/migrate.ts --schema",
    "db:migrate": "node -r ts-node/register src/migrate.ts",
    "typecheck": "tsc --noEmit"
  }
}
```

Create `tsconfig.json`:

```json
{
  "compilerOptions": {
    "module": "nodenext",
    "moduleResolution": "nodenext",
    "target": "ES2022",
    "types": ["node"],
    "strict": true,
    "esModuleInterop": true,
    "skipLibCheck": true
  },
  "include": ["src/**/*.ts", "crawlers/**/*.ts"]
}
```

## 2. Configure PostgreSQL and storage

For a local development database:

```sh
docker run --detach --name my-crawler-postgres \
  --publish 127.0.0.1:5433:5432 \
  --env POSTGRES_USER=admin \
  --env POSTGRES_PASSWORD=secret \
  --env POSTGRES_DB=my_crawler \
  postgres:17
```

Create `etc/config.json`:

```json
{
  "database": {
    "dialect": "postgres",
    "connection": {
      "host": "127.0.0.1",
      "port": 5433,
      "user": "admin",
      "password": "secret",
      "database": "my_crawler"
    }
  },
  "store": {
    "html": "data/html",
    "images": "data/images"
  },
  "sites": { "dir": "crawlers" }
}
```

Create a `.env` file for the server token; `runServer()` expects this file to exist.

```sh
node -e 'console.log("JSCRAWL_API_TOKEN=" + require("node:crypto").randomBytes(32).toString("hex"))' > .env
```

Exclude `.env`, `node_modules/`, `data/` and `*.log` from version control. You can
use `DATABASE_URL` instead of credentials in JSON; see [Configuration](configuration.md).

## 3. Define the page table and domain

Create `etc/pages.sql`. Core tables are loaded first, so this schema can reference
`site` and `url`.

```sql
DROP TABLE IF EXISTS page CASCADE;

CREATE TABLE page (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    site_id integer NOT NULL REFERENCES site(id) ON DELETE CASCADE,
    url_id bigint NOT NULL UNIQUE REFERENCES url(id) ON DELETE CASCADE,
    title text NOT NULL
);
```

Create `src/domain.ts`:

```ts
import { resolve } from 'node:path';

import type { XDocument } from 'gumbo-html';
import { Crawler } from 'jscrawl/base-crawler';
import { CoreDatabase } from 'jscrawl/base-database';
import { useDomain, type DomainConfig } from 'jscrawl/domain';
import type { SiteUrl } from 'jscrawl/models';
import { z } from 'zod';

export class PageDatabase extends CoreDatabase {
  addPage(url: SiteUrl, title: string) {
    return this.table('page').append({
      site: this.site.id,
      url: url.id,
      title,
    });
  }
}

export class PageCrawler extends Crawler<PageDatabase> {
  async scrapeSeed(doc: XDocument): Promise<void> {
    await this.scrapePage(doc);
  }

  async scrapePage(doc: XDocument): Promise<void> {
    const title = doc.first('title')?.text({ normalize: true }) ?? '';
    this.db.addPage(this.siteUrl, title);

    const current = new URL(this.response.url);
    for (const link of doc.find('a[href]')) {
      const href = link.attr('href');
      if (!href) continue;
      let target: URL;
      try {
        target = new URL(href, current);
      } catch {
        continue;
      }
      if (target.origin === current.origin) {
        this.db.addUrl(target.href, 'page', this.siteUrl.id);
      }
    }
  }
}

const config: DomainConfig = {
  name: 'pages',
  urlTypes: ['page'],
  schema: {
    files: [resolve(__dirname, '../etc/pages.sql')],
    migrations: [],
    version: 1,
  },
  database: PageDatabase,
  crawler: PageCrawler,
  expectations: {
    parse: (raw) => raw,
    shortfalls: () => [],
  },
  async metrics(siteId, db) {
    const [row] = await db.query<{ count: string }[]>(
      'select count(*) as count from page where site_id = ?',
      siteId,
    );
    return { pages: Number(row.count) };
  },
  metricRules: { pages: { label: 'pages' } },
  api: {
    resources: {
      Page: {
        filter: { fields: ['site', 'url'] },
        sort: { fields: ['id', 'title'], default: ['id'] },
      },
    },
  },
  inspectTool: 'inspectPage',
  tools: ({ db }) => [
    {
      name: 'inspectPage',
      readOnly: true,
      description: 'Read the saved title for a URL id.',
      schema: { url: z.string() },
      async execute({ url }) {
        const rows = await db.query('select title from page where url_id = ?', Number(url));
        return JSON.stringify(rows);
      },
    },
  ],
};

useDomain(config);
```

This deliberately small domain stores one title per URL. For production records,
add validation, expectations and deletion handling as described in
[Domains and records](domains.md).

## 4. Add the entry files and a site

Create the following four files. Each imports the domain before invoking the
library entry function.

`src/cli.ts`:

```ts
import './domain';
import { runCli } from 'jscrawl/entries/cli';

runCli().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

`src/migrate.ts`:

```ts
import './domain';
import { runMigrate } from 'jscrawl/entries/migrate';

runMigrate().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

`src/server.ts`:

```ts
import './domain';
import { runServer } from 'jscrawl/entries/server';

runServer().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

`src/mcp/server.ts`:

```ts
import '../domain';
import { runToolServer } from 'jscrawl/entries/tool-server';

runToolServer().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

The server starts children at `src/cli.ts` and `src/mcp/server.ts`; keep those
paths when using the standard server entry.

Create `crawlers/example.ts`. Replace the seed with a small site you intend to
crawl. The example follows links on the same origin; tailor discovery and path
rules to the pages you want before using it on a larger site.

```ts
import type { SiteConfig } from 'jscrawl/models';

import { PageCrawler } from '../src/domain';

export const SEED_URL = 'https://example.com/';
export const site: SiteConfig = {
  waitTime: 1,
  randomWait: 1,
  pageSnapshots: 'gzip',
  disallow: ['/search', '/login'],
};

export default class extends PageCrawler {}
```

## 5. Preview, then crawl

Create `crawlers/fixtures/example/home.html`:

```html
<!doctype html>
<html>
  <head>
    <title>My first page</title>
  </head>
  <body>
    Hello!
  </body>
</html>
```

Wait until PostgreSQL is accepting connections, then run:

```sh
npm run typecheck
npm run db:schema
npm run jscrawl -- --setup example
npm run jscrawl -- --scrape-file example seed crawlers/fixtures/example/home.html --url https://example.com/
npm run jscrawl -- --crawl example
```

**`db:schema` drops existing tables and data. Use it only for this new, disposable
database.** Use `db:migrate` for an existing installation; see
[Migrations](operations.md#database-migrations).

The preview prints buffered records with the title `My first page` and writes no
crawl records. It still needs the database schema. The crawl fetches the real
seed, saves its title, follows matching links and exits after the queue drains.
It may wait about 50 seconds at the end while checking for newly queued URLs.

## 6. Inspect the results

```sh
npm start
```

In another terminal in the app directory:

```sh
set -a
. ./.env
set +a
curl http://127.0.0.1:3120/healthz
curl -H "Authorization: Bearer $JSCRAWL_API_TOKEN" http://127.0.0.1:3120/api/overview
curl -H "Authorization: Bearer $JSCRAWL_API_TOKEN" \
  -H 'Content-Type: application/json' \
  --data '{"query":"select title from page order by id limit 10"}' \
  http://127.0.0.1:3120/api/sql
```

The health endpoint becomes ready when the MCP child is ready. `/api/overview`
shows queue and issue counts; the SQL example reads saved titles. The server
provides HTTP and MCP endpoints; it does not include a browser dashboard.

To fetch the site again after a completed crawl:

```sh
npm run jscrawl -- --reset example
npm run jscrawl -- --crawl example
```
